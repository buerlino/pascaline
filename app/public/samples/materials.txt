// An example of the library format (a few materials, two functions), not a material table: Pascaline keeps none.
// Copy its shape: each value exact and in SI, its unit in words in its one-line description, and above that
// its source as standard, edition and clause or table. "(to verify: owner)" marks a source not yet checked
// against the standard's text. Check every value against your own edition and grade before you rely on it.
// A name holds a plain number, in the unit view too: give it its unit where you use it (E_steel times Pa).
// The `// test:` lines run when the file is imported; their expected values follow from the values above them.

// source: EN 1993-1-1:2005, clause 3.2.6(1): E = 210 000 N/mm² (to verify: owner)
// Modulus of elasticity of structural steel, in Pa
E_steel:=21*10^10
// test: E_steel/10^6 == 210000

// source: EN 1993-1-2:2005, clause 3.2.2(1): unit mass 7850 kg/m³ (to verify: owner)
// Density of structural steel, in kg/m³
rho_steel:=7850

// source: EN 1993-1-1:2005, clause 3.2.6(1): α = 12×10⁻⁶ per K, for T up to 100 °C (to verify: owner)
// Coefficient of linear thermal expansion of structural steel, in 1/K
alpha_steel:=12*10^-6

// source: EN 1993-1-2:2005, clause 3.4.1.3(1): λ = 54 − 3.33×10⁻² θ, here at θ = 20 °C (to verify: owner)
// Thermal conductivity of structural steel at 20 °C, in W/(m·K)
lambda_steel:=53334*10^-3
// test: lambda_steel == 54 - 333/10000*20

// source: EN 1999-1-1:2007, clause 3.2.5(1): E = 70 000 N/mm² (to verify: owner)
// Modulus of elasticity of structural aluminium alloys, in Pa
E_alu:=7*10^10
// test: E_alu/10^6 == 70000

// source: EN 1999-1-1:2007, clause 3.2.5(1): ρ = 2700 kg/m³ (to verify: owner)
// Density of structural aluminium alloys, in kg/m³
rho_alu:=2700

// source: EN 1999-1-1:2007, clause 3.2.5(1): α = 23×10⁻⁶ per °C (to verify: owner)
// Coefficient of linear thermal expansion of structural aluminium alloys, in 1/K
alpha_alu:=23*10^-6

// source: EN 1999-1-2:2007, clause 3.3.1.3: λ = 0.07 θ + 190 for 3xxx and 6xxx alloys, at θ = 20 °C (to verify: owner)
// Thermal conductivity of a 6xxx aluminium alloy (EN AW-6082, say) at 20 °C, in W/(m·K)
lambda_alu:=1914*10^-1
// test: lambda_alu == 7/100*20 + 190

// source: Deutsches Kupferinstitut, data sheet Cu-ETP (CW004A), annealed: E = 127 kN/mm² (to verify: owner)
// Modulus of elasticity of copper Cu-ETP, annealed, in Pa
E_copper:=127*10^9
// test: E_copper/10^6 == 127000

// source: Deutsches Kupferinstitut, data sheet Cu-ETP (CW004A): 8.93 g/cm³ (to verify: owner)
// Density of copper Cu-ETP, in kg/m³
rho_copper:=8930

// source: Deutsches Kupferinstitut, data sheet Cu-ETP (CW004A): 17×10⁻⁶ per K, 20 to 100 °C (to verify: owner)
// Coefficient of linear thermal expansion of copper Cu-ETP, 20 to 100 °C, in 1/K
alpha_copper:=17*10^-6

// source: Deutsches Kupferinstitut, data sheet Cu-ETP (CW004A): 394 W/(m·K) at 20 °C (to verify: owner)
// Thermal conductivity of copper Cu-ETP at 20 °C, in W/(m·K)
lambda_copper:=394

// source: EN 1992-1-1:2004, Table 3.1: Ecm = 33 GPa for C30/37 (to verify: owner)
// Secant modulus of elasticity Ecm of concrete C30/37 (quartzite aggregate), in Pa
E_C30:=33*10^9
// test: E_C30/10^6 == 33000

// source: EN 1991-1-1:2002, Annex A, Table A.1: 24 kN/m³ for plain concrete, taken as 2400 kg/m³ (to verify: owner)
// Density of normal-weight plain concrete, in kg/m³
rho_C30:=2400

// source: EN 1992-1-1:2004, clause 3.1.3(5): α = 10×10⁻⁶ per K (to verify: owner)
// Coefficient of linear thermal expansion of normal-weight concrete, in 1/K
alpha_C30:=10*10^-6

// source: EN 1992-1-2:2004, clause 3.3.3(2), upper limit, at θ = 20 °C (to verify: owner)
// Thermal conductivity of normal-weight concrete at 20 °C (upper limit), in W/(m·K)
lambda_C30:=1951408*10^-6
// test: lambda_C30 == 2 - 2451/10000*(20/100) + 107/10000*(20/100)^2

// Free thermal strain ε = α·ΔT (no unit; α in 1/K, ΔT in K): use it as thermal_strain(alpha_steel,30)
thermal_strain(alpha,dT):=alpha*dT
// test: thermal_strain(alpha_steel,100) == 3/2500
// test: thermal_strain(alpha_alu,50) == 23/20000

// Axial stiffness k = E·A/L of a bar, in N/m (E in Pa, A in m², L in m)
axial_stiffness(E,A,L):=E*A/L
// test: axial_stiffness(E_steel,1/1000,2) == 105000000
// test: simplify(2*axial_stiffness(a,b,2*c) - axial_stiffness(a,b,c)) == 0
